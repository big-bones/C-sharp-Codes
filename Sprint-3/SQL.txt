create procedure addbooks(
	@book_name nvarchar(100),
	@author_id int,
	@book_price decimal,
	@is_available bit
)
as 
begin
	declare @last_id int
	select top 1 @last_id = bookid from books
	order by bookid desc;
	set @last_id = @last_id + 1
	insert into books(bookid,bookname,authorid,price,isavailable)
	values (@last_id,@book_name,@author_id,@book_price,@is_available)
end;
go
