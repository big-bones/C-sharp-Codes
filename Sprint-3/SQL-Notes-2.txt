from Employees e
where e.Salary > ( select AVG(salary) from Employees 
					group by Employees.DepartmentID 
					having Employees.DepartmentID = e.DepartmentID 
				)

--Also includes the employees that do not have a department
with cte_greater_than_average as(
	select EmployeeName,EmployeeID,salary,
	case
	when DepartmentID is null then 'No Department'
	else cast(DepartmentID as varchar(20))
	end as Department_Name,
	row_number() over(partition by departmentid order by salary desc) as row_num,
	avg(salary) over (partition by departmentid) as average_salary
	from employees
)select * from cte_greater_than_average
where salary >= average_salary

alter procedure sp_temp_table_example
(@DepartmentID int)
as
begin

	create table #employee_data(
		EmployeeID int,
		EmployeeName varchar(20),
		DepartmentID int,
		Salary decimal(10,2)
	);

	create clustered index temp_table_index
	on #employee_data(EmployeeID)

	insert into #employee_data(EmployeeID,EmployeeName,DepartmentID,Salary)
	select EmployeeID,EmployeeName,DepartmentID,Salary 
	from Employees where Employees.DepartmentID = @DepartmentID

	select * from #employee_data

	drop table #employee_data

end

alter procedure sp_change_salary(
	@employee_from_id int,
	@employee_to_id int,
	@salary_change decimal(10,2)
)
as
begin 
	declare @row_count_from_change int
	declare @row_count_to_change int
	declare @temp_storage decimal(10,2)
	begin try
	begin transaction
		select @temp_storage = salary from employees 
		where EmployeeID = @employee_from_id
		if @temp_storage >= @salary_change
		begin
			set nocount on;
			update employees
			set salary = salary - @salary_change
			where employeeID = @employee_from_id
			set @row_count_from_change = @@ROWCOUNT

			update employees
			set salary = salary + @salary_change
			where employeeID = @employee_to_id

			set @row_count_to_change = @@ROWCOUNT

			if @row_count_from_change = 1 and @row_count_to_change = 1
			begin 
				COMMIT TRANSACTION
				set nocount off
				print 'Success'
			end
			else
			begin
				RAISERROR('Wrong updates...',16,1)
			end

		end
		else
		begin
			RAISERROR('Insufficient Balance...',16,1)
		end
	end try
	begin catch 
		ROLLBACK TRANSACTION
		set nocount off
		print 'Error Occurred: ' + ERROR_MESSAGE()
	end catch
end

exec sp_change_salary 4,5,4000

--With CTE and Window Functions
with cte_delete_duplicates
as(
	select employeename,departmentid,employeeid,
	row_number() over(partition by employeename order by employeeid) as row_num
	from employees
)
delete from employees
where employeeid in (
	select employeeid
	from cte_delete_duplicates
	where employeename = 'A' 
	and row_num > 1
)


with cte_as_delete_all_duplicates_no_condition as(
	select employeeid
	row_number() over(partition by employeename order by employeeid) as row_num
	from employees
)
delete from employees
where employeeid in (
   select employeeid from 
   cte_as_delete_all_duplicates_no_condition
   where employeeid > 1
)

create trigger triggerAfterSalaryUpdate
on employees
after update
as 
begin
	insert into EmployeeAudit(EmployeeID,OldSalary,NewSalary)
	select 
		inserted.EmployeeID,
		deleted.Salary as OldSalary,
		inserted.Salary as NewSalary
	from inserted
	join deleted
	on inserted.EmployeeID = deleted.EmployeeID
end

--second highest salary

with cte_second_highest as(
	select EmployeeName,DepartmentID,Salary,
	ROW_NUMBER() over(partition by departmentid order by salary desc) as row_num,
	count(*) over(partition by departmentid) as dept_count
	from employees
)SELECT
    EmployeeName,
    DepartmentID,
    Salary,
    CASE
        WHEN dept_count = 1 THEN 'Only one employee in the department'
        ELSE 'Second highest salary'
    END AS Message
FROM cte_second_highest
WHERE row_num = 2 OR dept_count = 1;

ALTER PROCEDURE sp_employee_iteration
AS 
BEGIN
    DECLARE @EmployeeID INT;
    DECLARE @EmployeeName VARCHAR(20);
    DECLARE @DepartmentID INT;

    DECLARE iterationcursor CURSOR FOR
    SELECT EmployeeID, EmployeeName, DepartmentID
    FROM Employees;

    OPEN iterationcursor;

    FETCH NEXT FROM iterationcursor INTO @EmployeeID, @EmployeeName, @DepartmentID;

    WHILE @@FETCH_STATUS = 0
    BEGIN
        -- Use CONCAT or CAST/CONVERT to concatenate strings
        PRINT 'Employee Data ' + CAST(@EmployeeID AS VARCHAR(10)) + ' ' + @EmployeeName + ' ' + CAST(@DepartmentID AS VARCHAR(10));

        FETCH NEXT FROM iterationcursor INTO @EmployeeID, @EmployeeName, @DepartmentID;
    END

    CLOSE iterationcursor;
    DEALLOCATE iterationcursor;
END
