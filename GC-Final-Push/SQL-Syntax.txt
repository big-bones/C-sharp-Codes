create trigger after_update_trigger on books
after update 
as
begin
	select * from inserted join deleted
	on inserted.BookID = deleted.BookID
end
---------------------------------------------------------------
create trigger instead_of_update_trigger on books
instead of update
as
begin
	select * from inserted join deleted
	on inserted.BookID = deleted.BookID
end
